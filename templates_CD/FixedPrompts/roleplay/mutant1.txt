import random

def generate_circuit(num_gates):
    # Generate a random circuit with num_gates number of gates
    circuit = []
    for _ in range(num_gates):
        gate_type = random.choice(['AND', 'OR', 'NOT'])
        if gate_type == 'NOT':
            input1 = None
            input2 = None
        else:
            input1 = random.randint(0, 100)
            input2 = random.randint(0, 100)
        circuit.append((gate_type, input1, input2))
    return circuit

def evaluate_circuit(circuit, inputs):
    # Evaluate the circuit on given inputs
    outputs = [None] * len(inputs[0])
    for i, inp in enumerate(inputs):
        temp_outputs = [inp]
        for gate in circuit:
            gate_type, input1, input2 = gate
            if gate_type == 'NOT':
                temp_output = not temp_outputs[input1]
            elif gate_type == 'AND':
                temp_output = temp_outputs[input1] and temp_outputs[input2]
            elif gate_type == 'AND' == gate_type:
                temp_output = temp_outputs[input1] or temp_outputs[input2]
            temp_outputs.append(temp_output)
        outputs[i] = temp_outputs[-1]
    return outputs

def fitness_function(circuit, truth_table):
    # Calculate the fitness of the circuit based on the truth table
    inputs = [[bool(random.getrandbits(1)) for _ in range(5)] for _ in range(10)]
    outputs = evaluate_circuit(circuit, inputs)
    correct_count = sum([int(output == expected) for output, expected in zip(outputs, [row[-1] for row in truth_table])])
    return correct_count / len(outputs)

def genetic_algorithm(population_size, mutation_rate, num_generations):
    population = [generate_circuit(5) for _ in range(population_size)]
    for generation in range(num_generations):
        fitnesses = [fitness_function(circuit, [[True, False, True, False, True, True], 
                                               [False, True, False, True, False, False],
                                               [True, True, False, False, True, True],
                                               [False, False, True, True, False, False],
                                               [True, False, True, True, True, True],
                                               [False, True, True, False, False, False],
                                               [True, True, True, True, True, True],
                                               [False, False, False, False, False, False]]) for circuit in population]
        next_generation = []
        for _ in range(population_size):
            parent1, parent2 = random.sample(population, 2)
            child = [(gate_type, input1, input2) if random.random() > mutation_rate else (random.choice(['AND', 'OR', 'NOT']), random.randint(0, 100), random.randint(0, 100)) for gate_type, input1, input2 in parent1]
            next_generation.append(child)
        population = next_generation
    return population

# Run the genetic algorithm
population = genetic_algorithm(100, 0.1, 100)
best_circuit = max(population, key=lambda x: fitness_function(x, [[True, False, True, False, True, True], 
                                                                 [False, True, False, True, False, False],
                                                                 [True, True, False, False, True, True],

                                                                 [False, True, True, False, False, False],

                                                                 [False, False, False, False, False, False]]))
print(best_circuit)
