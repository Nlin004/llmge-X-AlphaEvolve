import jax.numpy as jnp

def cp_decomposition(matrix):
    # Initialize random weights and factors for each mode
    weights = jnp.array([0.5, 0.3, 0.2])
    factors = [jnp.random.rand(10, 1) for _ in range(3)]

    # Perform iterative updates using ALS (Alternating Least Squares)
    for _ in range(100):
        for i in range(3):
            # Compute Khatri-Rao product of all other modes
            kr_product = jnp.kron(jnp.kron(factors[0], factors[1]), factors[2]) if i == 0 else \
                         jnp.kron(jnp.kron(factors[0], factors[2]), factors[1]) if i == 1 else \
                         jnp.kron(jnp.kron(factors[1], factors[2]), factors[0])

            # Update factor for current mode
            factors[i] = jnp.dot(matrix, kr_product)

    return weights, factors

# Example usage
matrix = jnp.random.rand(10, 10, 10)
weights, factors = cp_decomposition(matrix)
print(weights, factors)
