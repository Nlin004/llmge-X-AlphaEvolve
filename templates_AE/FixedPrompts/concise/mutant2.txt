import numpy as np

def cp_decomposition(matrix, num_components):
    """
    Perform CP decomposition on the input matrix.

    Parameters:
        - matrix (numpy array): Input matrix to be decomposed.
        - num_components (int): Number of components for the decomposition.

    Returns:
        - A tuple containing the factor matrices.
    """
    # Initialize factor matrices randomly
    factor_matrices = [np.random.rand(matrix.shape[i], num_components) for i in range(2)]

    # Normalize the columns of each factor matrix
    factor_matrices = [fm / np.linalg.norm(fm, axis=0, keepdims=True) for fm in factor_matrices]

    return tuple(factor_matrices)
