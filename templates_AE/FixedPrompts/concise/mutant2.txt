import jax.numpy as jnp

def cp_decomposition(matrix):
    # Initialize random weights for each factor matrix
    num_factors = 3
    ranks = [2, 2, 2]
    weights = [jnp.random.rand(r) for r in ranks]

    # Perform iterative updates using gradient descent
    learning_rate = 0.01
    iterations = 1000
    for _ in range(iterations):
        # Compute the predicted tensor
        predicted_tensor = jnp.einsum('ij,jk,kl->ikl', weights[0], weights[1], weights[2])

        # Compute the loss (mean squared error)
        loss = jnp.mean((matrix - predicted_tensor) ** 2)

        # Backpropagate the gradients
        grads = jax.grad(loss, weights)

        # Update the weights using gradient descent
        weights = [w - learning_rate * g for w, g in zip(weights, grads)]

    return weights
