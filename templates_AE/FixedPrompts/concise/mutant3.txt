import jax.numpy as jnp

def cp_decomposition(matrix, num_terms):
    # Initialize random weights and factors for each rank-1 term
    weights = jnp.array([jnp.random.uniform(0, 1) for _ in range(num_terms)])
    factors = [jnp.random.uniform(0, 1, size=(matrix.shape[0],)) for _ in range(num_terms)]

    # Define the loss function (mean squared error)
    def loss(weights, factors):
        reconstruction = jnp.sum([w * jnp.outer(f, f) for w, f in zip(weights, factors)], axis=0)
        return jnp.mean((reconstruction - matrix) ** 2)

    # Perform gradient descent optimization
    learning_rate = 0.01
    iterations = 1000
    opt_state = jax.optimizers.init_optimize(
        [(weights, factors)],
        jax.optimizers.adam(learning_rate),
    )
    for i in range(iterations):
        grads = jax.grad(loss)(weights, factors)
        opt_state = jax.optimizers.apply_updates(opt_state, grads)

    return weights, factors

# Example usage
matrix = jnp.array([[1, 2], [3, 4]])
num_terms = 2
weights, factors = cp_decomposition(matrix, num_terms)
print("Weights:", weights)
print("Factors:", factors)
