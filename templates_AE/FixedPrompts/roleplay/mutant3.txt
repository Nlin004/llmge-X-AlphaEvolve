import jax.numpy as jnp

def loss_fn(params):
    # params is a list of 3 tensors representing the factors of the CP decomposition
    reconstructed_tensor = jnp.einsum('ir,jr,kr->ijk', params[0], params[1], params[2])
    target_tensor = jnp.array([...])  # placeholder for actual data
    return jnp.sum((reconstructed_tensor - target_tensor) ** 2)

# Initialize parameters
params_init = [jnp.random.rand(10, 5), jnp.random.rand(20, 5), jnp.random.rand(30, 5)]

# Define the optimizer
from jax.experimental import optimizers
opt_init, opt_update, get_params = optimizers.adam(step_size=0.01)
opt_state = opt_init(params_init)

# Training loop
for i in range(1000):
    grads = jax.grad(loss_fn)(get_params(opt_state))
    opt_state = opt_update(i, grads, opt_state)
