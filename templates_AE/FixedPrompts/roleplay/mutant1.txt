import jax.numpy as jnp

def cp_decomposition(matrix):
    # Initialize random weights
    weights = jnp.array([jnp.random.rand() for _ in range(10)])

    # Define the loss function
    def loss(weights):
        reconstructed_matrix = jnp.outer(weights, weights)
        return jnp.sum((matrix - reconstructed_matrix) ** 2)

    # Perform gradient descent optimization
    opt_state = jax.optimizers.init_optimize(
        weights,
        jax.optimizers.adam(0.01),
    )
    for i in range(1000):
        grads = jax.grad(loss)(opt_state[0])
        opt_state = jax.optimizers.apply_optimization(
            opt_state,
            grads,
            jax.optimizers.adam(0.01),
        )

    return opt_state[0]
