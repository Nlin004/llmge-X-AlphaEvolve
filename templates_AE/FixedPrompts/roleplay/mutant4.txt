import jax.numpy as jnp

def cp_decomposition(matrix):
    # Define the number of iterations and learning rate
    num_iterations = 1000
    learning_rate = 0.01
    
    # Initialize the weights randomly
    weights = jnp.random.uniform(0, 1, size=(matrix.shape[0],))
    
    # Perform gradient descent to optimize the weights
    for _ in range(num_iterations):
        # Compute the gradient of the loss function
        gradient = -2 * jnp.dot(matrix.T, (jnp.outer(weights, weights) - matrix))
        
        # Update the weights using gradient descent
        weights -= learning_rate * gradient
        
    return weights

# Example usage:
matrix = jnp.array([[1, 2], [3, 4]])
result = cp_decomposition(matrix)
print(result)
